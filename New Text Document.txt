    <script>
        // البيانات المالية
        const totalRevenue = 39600.00;
        const fixedExpenses = 15920.09;
        const variableExpenses = 4057.00;
        const eventExpenses = 4000.00; // مصروفات الحفلات
        const totalExpenses = fixedExpenses + variableExpenses + eventExpenses;
        const tax = 1980.00; // 5% من الإيرادات
        const netProfit = totalRevenue - totalExpenses - tax;

        // تحديث الملخص
        document.getElementById('totalRevenue').textContent = totalRevenue.toFixed(2) + " د.إ";
        document.getElementById('totalExpenses').textContent = totalExpenses.toFixed(2) + " د.إ";
        document.getElementById('tax').textContent = tax.toFixed(2) + " د.إ";
        document.getElementById('netProfit').textContent = netProfit.toFixed(2) + " د.إ";

        // إنشاء الرسوم البيانية
        const mainCtx = document.getElementById('mainChart').getContext('2d');
        const comparisonCtx = document.getElementById('comparisonChart').getContext('2d');

        let mainChart, comparisonChart;

        // رسم توزيع الإيرادات والمصروفات
        function renderRevenuesChart() {
            const data = {
                labels: ['إيرادات', 'مصروفات ثابتة', 'مصروفات متغيرة', 'مصروفات الحفلات'],
                datasets: [{
                    data: [totalRevenue, fixedExpenses, variableExpenses, eventExpenses],
                    backgroundColor: [
                        'rgba(75, 192, 192, 0.7)',
                        'rgba(255, 99, 132, 0.7)',
                        'rgba(255, 206, 86, 0.7)',
                        'rgba(153, 102, 255, 0.7)'
                    ],
                    borderWidth: 1
                }]
            };

            if (mainChart) mainChart.destroy();

            mainChart = new Chart(mainCtx, {
                type: 'pie',
                data: data,
                options: {
                    responsive: true,
                    plugins: {
                        legend: { position: 'right' },
                        datalabels: {
                            formatter: (value, ctx) => {
                                const total = ctx.chart.data.datasets[0].data.reduce((a, b) => a + b, 0);
                                const percentage = Math.round((value / total) * 100);
                                return `${percentage}%`;
                            },
                            color: getComputedStyle(document.body).getPropertyValue('--text-color'),
                            font: {
                                weight: 'bold'
                            }
                        }
                    }
                },
                plugins: [ChartDataLabels]
            });
        }

        // رسم توزيع المصروفات
        function renderExpensesChart() {
            const data = {
                labels: ['مصروفات ثابتة', 'مصروفات متغيرة', 'مصروفات الحفلات'],
                datasets: [{
                    data: [fixedExpenses, variableExpenses, eventExpenses],
                    backgroundColor: [
                        'rgba(255, 99, 132, 0.7)',
                        'rgba(255, 206, 86, 0.7)',
                        'rgba(153, 102, 255, 0.7)'
                    ],
                    borderWidth: 1
                }]
            };

            if (mainChart) mainChart.destroy();

            mainChart = new Chart(mainCtx, {
                type: 'pie',
                data: data,
                options: {
                    responsive: true,
                    plugins: {
                        legend: { position: 'right' },
                        datalabels: {
                            formatter: (value, ctx) => {
                                const total = ctx.chart.data.datasets[0].data.reduce((a, b) => a + b, 0);
                                const percentage = Math.round((value / total) * 100);
                                return `${percentage}%`;
                            },
                            color: getComputedStyle(document.body).getPropertyValue('--text-color'),
                            font: {
                                weight: 'bold'
                            }
                        }
                    }
                },
                plugins: [ChartDataLabels]
            });
        }

        // رسم مقارنة الإيرادات والمصروفات وصافي الربح
        function renderComparisonChart() {
            const data = {
                labels: ['إيرادات', 'مصروفات', 'ضريبة', 'صافي الربح'],
                datasets: [{
                    label: 'القيمة (د.إ)',
                    data: [totalRevenue, totalExpenses, tax, netProfit],
                    backgroundColor: [
                        'rgba(75, 192, 192, 0.7)',
                        'rgba(255, 99, 132, 0.7)',
                        'rgba(255, 206, 86, 0.7)',
                        'rgba(54, 162, 235, 0.7)'
                    ],
                    borderWidth: 1
                }]
            };

            if (comparisonChart) comparisonChart.destroy();

            comparisonChart = new Chart(comparisonCtx, {
                type: 'bar',
                data: data,
                options: {
                    responsive: true,
                    scales: {
                        y: { beginAtZero: true }
                    }
                }
            });
        }

        // تحديث الرسوم البيانية
        function updateCharts() {
            const dataType = document.getElementById('dataType').value;
            if (dataType === 'revenues') {
                document.getElementById('chartTitle').textContent = 'توزيع الإيرادات والمصروفات';
                renderRevenuesChart();
            } else {
                document.getElementById('chartTitle').textContent = 'توزيع المصروفات';
                renderExpensesChart();
            }
            renderComparisonChart();
        }

        // تبديل الوضع الداكن
        function toggleDarkMode() {
            document.body.classList.toggle('dark-mode');
            updateCharts();
        }

        // تهيئة الرسوم عند تحميل الصفحة
        window.onload = function() {
            updateCharts();
        };
